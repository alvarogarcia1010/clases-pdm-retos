  const [cantidad, setCantidad] = useState(0);
  return (
    <View>
      <Text>{cantidad}</Text>
      <Pressable onPress={() => setCantidad(cantidad + 1)}>
        <Text>Agregar</Text>
      </Pressable>
    </View>
